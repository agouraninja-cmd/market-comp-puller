# 1031 Exchange Guide (v4 slice 3) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A public education page at `GET /1031-exchange` — the 1031 workflow
in plain English with a client-side 45/180-day deadline-dates widget — usable
by owners researching a sale and by brokers as a client hand-out.

**Architecture:** All content lives in a new pure module `guide-1031.js`
(the vault-page.js precedent); `server.js` only wires a route through the
existing `marketShell()`, adds the sitemap line, and gains three inbound
links. Spec: `docs/superpowers/specs/2026-08-08-1031-guide-design.md`.

**Tech Stack:** Plain Node ≥18 built-ins, `node --test`, zero npm deps, no
build step, no migration, no database.

## Global Constraints

- **Education, never advice.** The page carries a visible box saying it is
  educational only — not tax, legal, or investment advice — and that
  CompNinja is not a brokerage, qualified intermediary, or tax advisor.
  Tests pin these strings and assert the absence of self-claims.
- **The widget computes DATES only** — no gains, taxes, or dollar figures.
  Client-side only; nothing is sent to any server, and the page says so.
- Valuation CTAs say **"automated estimate"**, never appraisal. Broker copy
  is **"connect you with a local broker"**, never that we are one.
- Zero npm dependencies. `index.html` untouched. No migration, no env var.
- `npm test` green before every commit. `node --check server.js` after
  touching it.
- **A parallel session (v4 slice 2) shares this checkout and pushes to the

  immediately before staging; stage explicit paths only; `devlog.json` (and

// Minimal DOM stub: just what the widget touches.
function runWidget(closingDateValue) {
  const els = {};
  const el = (id) => (els[id] = els[id] || {
    value: "", textContent: "", innerHTML: "", className: "",
    handlers: {}, addEventListener(ev, fn) { this.handlers[ev] = fn; },
  });
  const document = { getElementById: el };
  const body = G.renderGuide1031Body();
  new Function("document", scriptOf(body))(document);
  const input = els["q1031close"];
  assert.ok(input && input.handlers.input, "script must listen on #q1031close");
  input.value = closingDateValue;
  input.handlers.input();
  return els["q1031out"];
}

  const out = runWidget("2026-08-20");
  assert.match(out.innerHTML, /Oct 4, 2026/);   // +45 crosses a month
  assert.match(out.innerHTML, /Feb 16, 2027/);  // +180 crosses the year
});

test("45 and 180 day dates: leap year", () => {
  const out = runWidget("2027-12-17");
  assert.match(out.innerHTML, /Jan 31, 2028/);
  assert.match(out.innerHTML, /Jun 14, 2028/);  // counts Feb 29, 2028
});

test("an empty or invalid date clears the output rather than guessing", () => {
  assert.equal(runWidget("").innerHTML, "");
  assert.equal(runWidget("not-a-date").innerHTML, "");
});

function escGuide(s) {
  return String(s == null ? "" : s).replace(/[&<>"]/g, (c) =>
    ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
}

// The deadline widget's script. Calendar-day arithmetic only, done in local
// time from the typed Y-M-D so timezones can never shift a date. Month names
// are spelled here (not toLocaleDateString) so tests and visitors see the
// same string on every OS locale.
const WIDGET_JS = `(function(){
  var MONTHS=["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  var DAYS=["Sun","Mon","Tue","Wed","Thu","Fri","Sat"];
  var input=document.getElementById("q1031close");
  var out=document.getElementById("q1031out");
  function fmt(d){
    return DAYS[d.getDay()]+", "+MONTHS[d.getMonth()]+" "+d.getDate()+", "+d.getFullYear();
  }
  function update(){
    var m=/^(\\d{4})-(\\d{2})-(\\d{2})$/.exec(String(input.value||""));
    if(!m){ out.innerHTML=""; return; }
    var y=+m[1], mo=+m[2]-1, day=+m[3];
    var d45=new Date(y,mo,day+45), d180=new Date(y,mo,day+180);
    out.innerHTML="<span>Day 45 \\u2014 identify in writing by: "+fmt(d45)+"</span>"+
      "<span>Day 180 \\u2014 close your replacement by: "+fmt(d180)+"</span>";
  }
  input.addEventListener("input",update);
})();`;
